% Palatalization

0 -> ʲ / (l|d|n|s|t) _ (i|y|e|ø|æ)
0 -> ʲ / (z|s) _ i
s -> sʲ / _ tʲ


% Palatalize geminated consonants before some vowels
sː -> sʲː / _ (i|y|e|ø|æ)

dː -> dʲː / _ (i|y|e|ø|æ)
lː -> lʲː / _ i
zː -> zʲː / _ i

ʲ -> 0 / ʲ _

% I do not know why this might be. There are only two instances and both do this. (I didn't implement any exceptions rules if they only occurred once/inconsistently)
nʲ -> n / _ eʃ
